run_model <- function(dataset) {
  
  set.seed(123)
  
  n <- nrow(dataset)
  train_index <- sample(1:n, size = 0.8*n)
  
  train <- dataset[train_index, ]
  test  <- dataset[-train_index, ]
  
  control <- trainControl(method = "cv", number = 5)
  
  model <- train(
    FutureDist_5s ~ Speed + Acceleration + YawRate + Vel_X + Vel_Y,
    data = train,
    method = "lm",
    trControl = control
  )
  
  # Test predictions
  pred <- predict(model, newdata = test)
  
  # RMSE
  rmse <- sqrt(mean((pred - test$FutureDist_5s)^2))
  
  # R-squared
  rss <- sum((pred - test$FutureDist_5s)^2)
  tss <- sum((test$FutureDist_5s - mean(test$FutureDist_5s))^2)
  r2 <- 1 - rss/tss
  
  return(list(
    RMSE = rmse,
    R2 = r2,
    CV_Results = model$results
  ))
}
